% ═══ RESEARCH EXPERIENCE ═══
\section{Research Experience}

\textbf{AViD Journal — Automated Verification Pipeline} \hfill 2024--Present\\
\textit{Primary research project · Independent}

\begin{itemize}
  \item Designed an end-to-end pipeline for LLM-assisted autoformalization of mathematical papers in \textbf{Lean 4}.
  \item Developed a \textbf{novelty detection} system that checks formalized theorems against Mathlib along three axes: existence (has this been proved before?), non-triviality (does it differ from known results?), and structural proof-distance.
  \item Core finding: \textit{fidelity $\neq$ compilation} — LLM-generated formalizations can compile and still fail to preserve the logical meaning of the original theorem.
  \item Integrated \textbf{arXiv} and \textbf{TheoremSearch} for paper ingestion and metadata extraction.
  \item Built a modular Python pipeline with FastAPI backend, SSE streaming, and Claude API integration.
\end{itemize}

\vspace{0.3em}

\textit{Fellow Researcher · International collaboration}
